(3) \mathcal{E}_{R}(U)=U+\sqrt{\epsilon_{r}}\,(U-\mathbb{I})\,, \tag{3}
(4) \mathcal{E}_{R}(U)^{\dagger}\mathcal{E}_{R}(U)=(1-\epsilon_{r})U^{\dagger}U+O(\epsilon_{r}). \tag{4}
(5) \mathcal{E}_{C}(U)=\mathcal{E}_{S}(\mathcal{E}_{R}(U))=\sum_{k=0}^{m-1}\epsilon_{k}\mathcal{E}_{R}(U)^{k}. \tag{5}
(6) \mathcal{E}_{C}(U)^{\dagger}\mathcal{E}_{C}(U)\approx\left(\begin{array}[]{*{20}{c}}1&-\delta\\ -\delta&1\end{array}\right), \tag{6}
(7) \mathcal{E}_{C}(U)^{\dagger}\mathcal{E}_{C}(U)\approx\left(\begin{array}[]{*{20}{c}}1&-\delta\\ -\delta&1\end{array}\right)+\frac{1}{2}\left(\begin{array}[]{*{20}{c}}0&-i\gamma\\ i\gamma&0\end{array}\right). \tag{7}
