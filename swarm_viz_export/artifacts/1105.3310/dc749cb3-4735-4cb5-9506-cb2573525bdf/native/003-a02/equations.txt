(1) \mathcal{P}_{d} = \{p_{x}\colon x\in F^{n}, p_{x}(y)=p(y+x)\}. \tag{1}
(2) \mathcal{V}=\mathrm{span}(\mathcal{P}_{d}). \tag{2}
(3) \mathcal{C}_{d}=\mathrm{span}\{p_{x}\colon x\in V\}, \tag{3}
(4) \mathcal{C}_{d}=\mathrm{span}\{\chi_{A}\colon A\subseteq[n],|A|\leq d\}. \tag{4}
(5) \mathcal{C}_{d}=\mathrm{span}\{\chi_{A}:A\subseteq[n],\deg(\chi_{A})\leq d\} \tag{5}
(6) \mathcal{C}_{d}=\mathrm{span}\{\chi_{A}:\deg(\chi_{A})=d\} \oplus \mathcal{C}_{d-1}. \tag{6}
(7) \mathcal{C}_{d}=\mathrm{span}\{\chi_{A}:|A|=d\} \oplus \mathcal{C}_{d-1}. \tag{7}
(8) \mathcal{C}_{d}=\mathrm{span}\{\chi_{A}:|A|=d\}\oplus\cdots\oplus\mathrm{span}\{\chi_{A}:|A|=0\}. \tag{8}
