(1) H = -J\sum_{\langle i,j\rangle}\left(\hat{a}_{i}^{\dagger}\hat{a}_{j}+\hat{a}_{j}^{\dagger}\hat{a}_{i}\right)+U\sum_{i}\hat{n}_{i}(\hat{n}_{i}-1). \tag{1}
(2) J=4\pi a^{3}/m,\quad U=2\pi a^{3}/\lambda. \tag{2}
(3) \mathcal{H}=\bigoplus_{k=0}^{N-1}\mathcal{H}_{k}, \tag{3}
(4) \mathcal{H}_{k}=V_{k}\otimes|k\rangle\langle k|, \tag{4}
(5) V_{k}=\frac{1}{\sqrt{(N-k)(k+1)}}\sum_{l=-k}^{k}\hat{c}_{l}|l\rangle\langle l|. \tag{5}
(6) |\psi_{0}\rangle=\frac{1}{\sqrt{N}}\sum_{k=0}^{N-1}|k\rangle, \tag{6}
(7) \hat{S}_{\pm}=\sum_{k=0}^{N-1}\hat{c}_{\pm}|k\rangle\langle k|\otimes|k\rangle\langle k|, \tag{7}
(8) \hat{S}_{+}=\sum_{k=0}^{N-1}\hat{c}_{+}e^{-i\theta_{k}}|k\rangle\langle k|\otimes|k\rangle\langle k|+e^{-i\phi_{k}}|k\rangle\langle k|\otimes|k\rangle\langle k|. \tag{8}
