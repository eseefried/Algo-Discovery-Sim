(1) \hat{f}_{k}(x)=\sum_{i=0}^{2^{k}-1}a_{i}\chi_{[i/2^{k}, (i+1)/2^{k})}(x), \tag{1}
(2) \tilde{\hat{f}}_{k}(x)=\frac{1}{2^{k}}\sum_{i=0}^{2^{k}-1}a_{i}\chi_{[i/2^{k}, (i+1)/2^{k})}(x). \tag{2}
(3) |f(x)-\hat{f}_{k}(x)|=\left|\sum_{i=0}^{2^{k}-1}a_{i}\left(\chi_{[i/2^{k}, (i+1)/2^{k})}(x)-a_{i}\right)\right| \tag{3}
(4) =|f(x)-\tilde{\hat{f}}_{k}(x)|\leq\max_{y}|f(y)-\tilde{\hat{f}}_{k}(y)|. \tag{4}
(5) \max_{y}|f(y)-\tilde{\hat{f}}_{k}(y)| =\max_{j=0}^{2^{k}-1}|\int_{(j-1)/2^{k}}^{j/2^{k}}f(x)dx-\int_{(j-1)/2^{k}}^{j/2^{k}}\tilde{\hat{f}}_{k}(x)dx|. \tag{5}
(6) \leq\max_{j=0}^{2^{k}-1}\int_{(j-1)/2^{k}}^{j/2^{k}}|f(x)-\tilde{\hat{f}}_{k}(x)|dx. \tag{6}
(7) \leq\max_{j=0}^{2^{k}-1}\int_{(j-1)/2^{k}}^{j/2^{k}}|f(x)-\bar{f}(x)|dx. \tag{7}
(8) \leq\max_{j=0}^{2^{k}-1}\int_{(j-1)/2^{k}}^{j/2^{k}}\epsilon\,dx=2^{-k}\epsilon. \tag{8}
