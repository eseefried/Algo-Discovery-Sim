(1) \mathcal{C}_{A}(U)=\sum_{x\in A} U_{x}\otimes I_{\bar{x}}, \tag{1}
(2) \mathcal{C}_{B}(V)=I_{B^{\complement}}\otimes V, \tag{2}
(3) \mathcal{C}_{AB}(W)=\mathcal{C}_{A}(\mathcal{C}_{B}(W)), \tag{3}
(4) \mathcal{C}_{BA}(Z)=\mathcal{C}_{B}(\mathcal{C}_{A}(Z))=\sum_{y\in B} I_{y^{\complement}}\otimes\left(\sum_{x\in A} Z_{xy}\otimes I_{(xy)^{\complement}}\right). \tag{4}
(5) \mathcal{C}_{AB}(Z)=\mathcal{C}_{A}(\mathcal{C}_{B}(Z))=\sum_{x\in A}\sum_{y\in B} Z_{xy}\otimes I_{(xy)^{\complement}}. \tag{5}
(6) \mathcal{C}_{A}(Z)\mathcal{C}_{B}(Z)=\sum_{x\in A}\sum_{y\in B} Z_{xy}\otimes I_{(xy)^{\complement}} = \mathcal{C}_{AB}(Z). \tag{6}
(7) \mathcal{C}_{A}(Z)\mathcal{C}_{B}(Z)=\sum_{x\in A}\sum_{y\in B} Z_{xy}\otimes I_{(xy)^{\complement}}=\mathcal{C}_{AB}(Z). \tag{7}
(8) \mathcal{C}_{A}(Z)\mathcal{C}_{B}(Z)=\sum_{x\in A}\sum_{y\in B} Z_{xy}\otimes I_{(xy)^{\complement}}=Z_{AB}. \tag{8}
