(4) H_{\mathrm{ZZ}}=\sum_{i<j}J^{\prime}_{ij}\sigma^{z}_{i}\sigma^{z}_{j}. \tag{4}

(6) |\psi(t)\rangle=\exp\left(-i\int_{0}^{t}H(t')\,dt'\right)|\psi(0)\rangle. \tag{6}
